    <script>
        let CLAUDE_API_KEY = localStorage.getItem('claude_api_key') || '';
        let fullWorkbook = null;
        let currentSheetName = "";
        let originalData = null;
        let proposedData = null;
        let contextInstructions = "";

        // UI Helpers
        function showPage(p) { document.querySelectorAll('.page').forEach(pg => pg.classList.remove('active')); document.getElementById(p + 'Page').classList.add('active'); }

        // Stars & Setup [Existing logic]
        const canvas = document.getElementById('stars-canvas');
        const ctx = canvas.getContext('2d');
        canvas.width = window.innerWidth; canvas.height = window.innerHeight;

        // API Setup
        function saveAPIKey() { const k = document.getElementById('apiKeyInput').value; if(k.startsWith('sk-ant-')) { localStorage.setItem('claude_api_key', k); CLAUDE_API_KEY = k; alert('Saved!'); } }

        // Handle Spreadsheet Upload
        document.getElementById('uploadZone').onclick = () => document.getElementById('fileInput').click();
        document.getElementById('fileInput').onchange = (e) => {
            const file = e.target.files[0];
            const reader = new FileReader();
            reader.onload = (evt) => {
                const data = new Uint8Array(evt.target.result);
                fullWorkbook = XLSX.read(data, { type: 'array' });
                createSheetTabs();
                loadSheet(fullWorkbook.SheetNames[0]);
                document.getElementById('uploadZone').style.display = 'none';
                document.getElementById('excelPreview').classList.add('active');
            };
            reader.readAsArrayBuffer(file);
        };

        function createSheetTabs() {
            const container = document.getElementById('sheetTabs');
            container.innerHTML = '';
            fullWorkbook.SheetNames.forEach(name => {
                const btn = document.createElement('div');
                btn.className = 'sheet-tab';
                btn.innerText = name;
                btn.onclick = () => loadSheet(name);
                container.appendChild(btn);
            });
        }

        function loadSheet(name) {
            currentSheetName = name;
            document.querySelectorAll('.sheet-tab').forEach(t => t.classList.toggle('active', t.innerText === name));
            const sheet = fullWorkbook.Sheets[name];
            originalData = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: '' });
            proposedData = JSON.parse(JSON.stringify(originalData));
            renderTables();
        }

        function renderTables() {
            renderToTable('excelTableOriginal', originalData);
            renderToTable('excelTableProposed', proposedData, true);
        }

        function renderToTable(tableId, data, highlight = false) {
            const table = document.getElementById(tableId);
            table.innerHTML = '';
            data.forEach((row, rIdx) => {
                const tr = document.createElement('tr');
                row.forEach((cell, cIdx) => {
                    const td = document.createElement(rIdx === 0 ? 'th' : 'td');
                    td.innerText = cell;
                    if (highlight && rIdx > 0 && cell !== originalData[rIdx][cIdx]) {
                        td.className = 'cell-changed';
                    }
                    tr.appendChild(td);
                });
                table.appendChild(tr);
            });
        }

        // Instruction Ingestion
        async function handleContextFile(input) {
            const file = input.files[0];
            if (!file) return;
            contextInstructions = await file.text();
            document.getElementById('contextStatus').innerText = `✅ Attached: ${file.name} (Context Loaded)`;
        }

        async function sendMessage() {
            const input = document.getElementById('chatInput');
            const msg = input.value.trim();
            if (!msg || !originalData) return;

            addMsg('user', msg);
            input.value = '';

            const prompt = `Spreadsheet Context:
Headers: ${originalData[0].join(', ')}
Sample Data: ${JSON.stringify(originalData.slice(1, 5))}
Global Instructions: ${contextInstructions}

Task: Transform 'processedData' (2D array). 
User Command: ${msg}

Rules:
- Row 0 is headers.
- Return ONLY JS code.
- End with // SUMMARY: [text]`;

            try {
                const response = await fetch('https://api.anthropic.com/v1/messages', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'x-api-key': CLAUDE_API_KEY,
                        'anthropic-version': '2023-06-01',
                        'anthropic-dangerous-direct-browser-access': 'true'
                    },
                    body: JSON.stringify({
                        model: 'claude-3-sonnet-20240229',
                        max_tokens: 2000,
                        messages: [{ role: 'user', content: prompt }]
                    })
                });

                const res = await response.json();
                let code = res.content[0].text.replace(/```javascript|```/g, '').trim();
                
                let processedData = JSON.parse(JSON.stringify(proposedData));
                eval(code);
                proposedData = processedData;
                
                renderTables();
                document.getElementById('downloadBtn').disabled = false;
                addMsg('ai', "Changes applied to the proposed view. Highlighted in teal.");
            } catch (e) {
                addMsg('ai', "Error: " + e.message);
            }
        }

        function addMsg(role, text) {
            const div = document.createElement('div');
            div.className = `message ${role}`;
            div.innerText = text;
            document.getElementById('chatMessages').appendChild(div);
        }

        function downloadCleaned() {
            const ws = XLSX.utils.aoa_to_sheet(proposedData);
            const wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "Cleaned");
            XLSX.writeFile(wb, "cleaned_data.xlsx");
        }
